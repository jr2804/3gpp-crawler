    ```python
    """Workspace export to external wiki app formats.

    Generates file structures compatible with:
    - atomicmemory/llm-wiki-compiler (sources/ folder + optional .llmwiki/schema.json)
    - lucasastorian/llmwiki (research folder + wiki/ scaffold)
    """

    from __future__ import annotations

    from dataclasses import dataclass, field
    from pathlib import Path
    from typing import Literal

    ExportFormat = Literal["llm-wiki-compiler", "llmwiki"]


    @dataclass
    class ExportManifest:
        """Result of an export operation."""
        target_dir: Path
        format: ExportFormat
        files_written: list[Path] = field(default_factory=list)
        errors: list[str] = field(default_factory=list)


    def export_workspace(
        workspace_path: Path,
        target_dir: Path,
        format: ExportFormat,
        *,
        overwrite: bool = False,
    ) -> ExportManifest:
        """Export a workspace to the specified external wiki app format.

        Args:
            workspace_path: Path to the workspace (.ai/ folder with extraction artifacts).
            target_dir: Output directory for the exported file structure.
            format: Target external app format.
            overwrite: If True, overwrite existing files in target_dir.

        Returns:
            ExportManifest with files written and any errors.
        """
        if format == "llm-wiki-compiler":
            return _export_llm_wiki_compiler(workspace_path, target_dir, overwrite=overwrite)
        elif format == "llmwiki":
            return _export_llmwiki(workspace_path, target_dir, overwrite=overwrite)
        else:
            msg = f"Unknown export format: {format}"
            raise ValueError(msg)


    def _export_llm_wiki_compiler(
        workspace_path: Path,
        target_dir: Path,
        *,
        overwrite: bool = False,
    ) -> ExportManifest:
        """Export to atomicmemory/llm-wiki-compiler format.

        Creates:
          sources/          - Markdown/text files from extraction artifacts
          .llmwiki/
            schema.json     - Optional schema seed for page kinds
        """
        manifest = ExportManifest(target_dir=target_dir, format="llm-wiki-compiler")
        sources_dir = target_dir / "sources"
        llmwiki_dir = target_dir / ".llmwiki"

        sources_dir.mkdir(parents=True, exist_ok=True)
        llmwiki_dir.mkdir(parents=True, exist_ok=True)

        # Walk workspace .ai/ artifacts and copy as sources
        ai_dir = workspace_path / ".ai"
        if ai_dir.is_dir():
            for item in sorted(ai_dir.iterdir()):
                if item.suffix in (".md", ".json", ".txt"):
                    dest = sources_dir / item.name
                    if not dest.exists() or overwrite:
                        dest.write_text(item.read_text(encoding="utf-8"), encoding="utf-8")
                        manifest.files_written.append(dest)

        # Write schema.json seed
        schema = sources_dir / "schema.json"
        if not schema.exists() or overwrite:
            schema.write_text(
                '{\n  "pageKinds": ["concept", "entity", "comparison", "overview"],\n'
                '  "defaultKind": "concept"\n}\n',
                encoding="utf-8",
            )
            manifest.files_written.append(schema)

        return manifest


    def _export_llmwiki(
        workspace_path: Path,
        target_dir: Path,
        *,
        overwrite: bool = False,
    ) -> ExportManifest:
        """Export to lucasastorian/llmwiki format.

        Creates:
          wiki/             - Scaffolded wiki directory
          .llmwiki/
            index.db        - Placeholder for SQLite index
        """
        manifest = ExportManifest(target_dir=target_dir, format="llmwiki")
        wiki_dir = target_dir / "wiki"
        llmwiki_dir = target_dir / ".llmwiki"

        wiki_dir.mkdir(parents=True, exist_ok=True)
        llmwiki_dir.mkdir(parents=True, exist_ok=True)

        # Copy extraction markdown into wiki/
        ai_dir = workspace_path / ".ai"
        if ai_dir.is_dir():
            for item in sorted(ai_dir.iterdir()):
                if item.suffix in (".md", ".txt"):
                    dest = wiki_dir / item.name
                    if not dest.exists() or overwrite:
                        dest.write_text(item.read_text(encoding="utf-8"), encoding="utf-8")
                        manifest.files_written.append(dest)

        # Create placeholder index.db marker
        index_marker = llmwiki_dir / "index.db"
        if not index_marker.exists() or overwrite:
            index_marker.write_text("", encoding="utf-8")
            manifest.files_written.append(index_marker)

        return manifest


    __all__ = ["ExportFormat", "ExportManifest", "export_workspace"]
    ```

    Key design decisions:
    - Export is purely filesystem-based: read artifacts, write target structure.
    - No LLM calls, no embedding, no wiki compilation logic.
    - Each format generator is a separate private function for clarity.
    - Deterministic: same inputs produce same output (sorted iteration, no randomness).

    @workspace_app.command("export", help="Export workspace to external wiki app format.")
    def workspace_export(
        workspace_name: WorkspaceNameArgument,
        target: ExportTargetOption,
        format: ExportFormatOption = "llm-wiki-compiler",
        overwrite: ExportOverwriteOption = False,
    ) -> None:
        \"\"\"Export workspace artifacts to a format compatible with an external wiki app.\"\"\"
        from threegpp_ai.operations.workspace_export import export_workspace

        normalized = resolve_workspace_name(workspace_name)
        workspace_path = get_workspace_path(normalized)

        if not workspace_path.is_dir():
            console.print(f"[red]Workspace '{normalized}' not found at {workspace_path}[/red]")
            raise typer.Exit(1)

        with console.status(f"[cyan]Exporting to {format} format..."):
            result = export_workspace(
                workspace_path=workspace_path,
                target_dir=target,
                format=format,  # type: ignore[arg-type]
                overwrite=overwrite,
            )

        if result.errors:
            for err in result.errors:
                console.print(f"[red]Error: {err}[/red]")
            raise typer.Exit(1)

        console.print(f"[green]Exported {len(result.files_written)} files to {target}[/green]")
        console.print(f"[dim]Format: {format}[/dim]")
        for f in result.files_written:
            console.print(f"  [dim]{f}[/dim]")
    ```

    """Tests for workspace export to external wiki app formats."""

    from __future__ import annotations

    import tempfile
    from pathlib import Path

    from threegpp_ai.operations.workspace_export import export_workspace


    def _create_fake_workspace(path: Path) -> None:
        """Create a minimal workspace with .ai/ artifacts for testing."""
        ai_dir = path / ".ai"
        ai_dir.mkdir(parents=True)
        (ai_dir / "doc1.md").write_text("# Document 1\n\nContent.", encoding="utf-8")
        (ai_dir / "doc2.md").write_text("# Document 2\n\nMore content.", encoding="utf-8")
        (ai_dir / "manifest.json").write_text('{"version": 1}', encoding="utf-8")


    def test_export_llm_wiki_compiler_creates_sources_folder() -> None:
        with tempfile.TemporaryDirectory() as tmp:
            workspace = Path(tmp) / "workspace"
            workspace.mkdir()
            _create_fake_workspace(workspace)

            target = Path(tmp) / "export"
            result = export_workspace(workspace, target, format="llm-wiki-compiler")

            assert (target / "sources").is_dir()
            assert (target / ".llmwiki").is_dir()
            assert (target / ".llmwiki" / "schema.json").exists()
            assert len(result.files_written) >= 3  # doc1.md, doc2.md, manifest.json, schema.json


    def test_export_llmwiki_creates_wiki_folder() -> None:
        with tempfile.TemporaryDirectory() as tmp:
            workspace = Path(tmp) / "workspace"
            workspace.mkdir()
            _create_fake_workspace(workspace)

            target = Path(tmp) / "export"
            result = export_workspace(workspace, target, format="llmwiki")

            assert (target / "wiki").is_dir()
            assert (target / ".llmwiki").is_dir()
            assert (target / ".llmwiki" / "index.db").exists()
            assert len(result.files_written) >= 3


    def test_export_unknown_format_raises_error() -> None:
        with tempfile.TemporaryDirectory() as tmp:
            workspace = Path(tmp) / "workspace"
            workspace.mkdir()
            _create_fake_workspace(workspace)

            import pytest

            with pytest.raises(ValueError, match="Unknown export format"):
                export_workspace(workspace, Path(tmp) / "out", format="invalid-format")  # type: ignore[arg-type]


    def test_export_empty_workspace_creates_structure() -> None:
        with tempfile.TemporaryDirectory() as tmp:
            workspace = Path(tmp) / "workspace"
            workspace.mkdir()
            (workspace / ".ai").mkdir()

            target = Path(tmp) / "export"
            result = export_workspace(workspace, target, format="llm-wiki-compiler")

            assert (target / "sources").is_dir()
            assert (target / ".llmwiki" / "schema.json").exists()
            # Only schema.json written (no artifacts to copy)
            assert len(result.files_written) >= 1
    ```
